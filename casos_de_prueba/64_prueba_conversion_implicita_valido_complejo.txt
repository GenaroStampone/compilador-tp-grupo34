class evaluador_financiero
begin
    doublef saldo_acumulado;
    doublef registrar_interes(doublef tasa, ushortint periodo)
    ushortint constante_base;
    begin
        constante_base := 2$us;
        saldo_acumulado := saldo_acumulado + (tasa * periodo) / constante_base;
        if (saldo_acumulado > 100.0d+0)
            ret (saldo_acumulado);
        else
            ret (constante_base);
        end_if;
    end;
end;

doublef function procesar_metricas(doublef limite_max, doublef factor_escalar)
ushortint contador_entero;
doublef acumulador;
begin
    contador_entero := 4$us;
    acumulador := 10.5d+0;
    acumulador := (limite_max + contador_entero) * (20$us - factor_escalar);
    if (acumulador >= contador_entero)
        limite_max := (acumulador = contador_entero + factor_escalar);
    end_if;
    ret (contador_entero + 1$us);
end;

evaluador_financiero mi_evaluador;
ushortint entero_base, contador_iteraciones;
doublef res_doble, tasa_anual;

begin
    entero_base := 10$us;

    tasa_anual := 0.5d+0;

    res_doble := mi_evaluador.registrar_interes(entero_base, 3$us);

end;
