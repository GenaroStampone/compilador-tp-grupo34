class ControladorSistema begin
    {{ Tema 30: Herencia múltiple mediante extends }}
    extends DispositivoBase, ModuloAuxiliar;

    {{ Tema 27: Modificador private en atributos }}
    private ushortint claveSeguridad;
    private doublef ajusteInterno;

    {{ Tema 27: Declaración de clases y funciones amigas (friend) }}

    ushortint function consultarClave(ushortint salt) begin
        ret (claveSeguridad + salt);
    end;

    private doublef function calcularAjuste(doublef delta)
        doublef factorLocal;
    begin
        factorLocal := delta;
        ret (ajusteInterno * factorLocal);
    end;
end;

    ushortint acumulador;
begin
    acumulador := v1 + v2 + v3;

    doublef convertido;
begin
    {{ Tema 33 y Tema 34: Conversiones Explícitas }}

    ret (convertido * 1.25d-2);
