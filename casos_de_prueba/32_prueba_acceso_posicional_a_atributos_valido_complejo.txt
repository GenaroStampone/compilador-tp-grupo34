class vector_bidimensional 
    begin 
        doublef componente_x; 
        doublef componente_y; 
        doublef calcular_magnitud (doublef escala) 
            begin 
                ret ((componente_x + componente_y) * escala); 
            end; 
    end; 

ushortint function procesar_vectores (ushortint modo, doublef factor) 
    ushortint resultado_operacion; 
    begin 
        if (factor >= 1.0d+0) 
            resultado_operacion := modo; 
        else 
            resultado_operacion := 0$us; 
        end_if; 
        ret (resultado_operacion); 
    end;
